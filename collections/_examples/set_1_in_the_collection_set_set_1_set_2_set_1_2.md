title: '$\setdef{1}$, in the collection $\setdef{\setdef{1}, \setdef{2}, \setdef{1, 2}}$ ordered by $\subseteq$'
level: high-school
is: [minimal_element]
is_not: [minimum]
why:
  minimal_element: >-
    $\setdef{1}$ is a minimal element because no set in the collection is strictly contained in it.
  minimum: >-
    $\setdef{1}$ is not the minimum, because $\setdef{1} \not\subseteq \setdef{2}$.
