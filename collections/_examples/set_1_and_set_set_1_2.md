title: '$\setdef{1}$ and $\setdef{\setdef{1}, 2}$'
level: elementary
is: [element_of]
is_not: [
  subset,
  proper_subset
]
why:
  element_of: >-
    $\setdef{1}$ is an element of $\setdef{\setdef{1}, 2}$ because it is listed in the set.
  subset: >-
    $\setdef{1}$ and $\setdef{\setdef{1}, 2}$ is not a subset, because its element $1$ is not an element of $\setdef{\setdef{1}, 2}$.
---
