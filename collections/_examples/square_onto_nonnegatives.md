title: '$f(x) = x^2$, as a function $f : \reals \to [0, \infty)$'
level: high-school
is: [surjective]
is_not: [injective]
values:
  domain: '$\reals$'
  codomain: '$[0, \infty)$'
  image: '$f(\reals) = [0, \infty)$'
  preimage: '$f^{-1}(\setdef{4}) = \setdef{-2, 2}$'
why:
  surjective: >-
    Every nonnegative number is the square of a real number.
  injective: >-
    $f(-2) = f(2)$.
  codomain: >-
    Now the image equals the codomain, so $f$ is surjective.
