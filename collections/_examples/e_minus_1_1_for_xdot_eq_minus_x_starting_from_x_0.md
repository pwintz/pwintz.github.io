published: false
title: '$[e^{-1}, 1]$, for $\dot{x} = -x$, starting from $X_0 = \setdef{1}$, up to time $1$'
level: high-school
is: [reachable_set]
is_not: [forward_invariant]
why:
  reachable_set: >-
    $[e^{-1}, 1]$ is the reachable set because it contains the states $e^{-t}$ for $0 \leq t \leq 1$.
  forward_invariant: >-
    $[e^{-1}, 1]$ is not forward invariant, because solutions keep decreasing below $e^{-1}$.
