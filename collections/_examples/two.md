title: '$2$'
level: high-school
is: [
  integer,
  rational_number,
  algebraic_number,
  real_number
]
is_not: [
  irrational_number,
  transcendental_number
]
why:
  rational_number: >-
    $2$ is an integer, and so a rational number ($2/1$) and a real number.
  algebraic_number: >-
    $2$ is algebraic, because it is a root of $x - 2$, so it is not irrational or transcendental.
