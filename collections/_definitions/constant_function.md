title: Constant function
term: constant
applies_to: functions
levels:
  elementary:
    text: |-
      A _constant_ function always gives the same output, no matter the input.
  high-school:
    text: |-
      A function is _constant_ if $f(x) = c$ for every $x$ in its domain, for some fixed $c$.
